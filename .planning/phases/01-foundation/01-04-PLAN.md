**main.go** — update `cmd/sops-tui/main.go`:

```go
package main

import (
    "fmt"
    "os"
    "strings"

    tea "charm.land/bubbletea/v2"
    "github.com/caesarakalaeii/sops-tui/internal/app"
    "github.com/caesarakalaeii/sops-tui/internal/ui"
    "github.com/caesarakalaeii/sops-tui/internal/validator"
)

func main() {
    // Step 1: Run all startup validation (before any TUI — per D-01, Pitfall 3)
    results, hasHardError := validator.RunChecks()

    // Step 2: If any issues, render styled error box to stderr (per D-01, D-02)
    if len(results) > 0 {
        ui.RenderErrorBox(results, hasHardError, os.Stderr)
    }

    // Step 3: Hard error = exit (per D-03)
    if hasHardError {
        os.Exit(1)
    }

    // Step 4: Build env status from validation results for status bar
    env := ui.EnvStatus{
        SopsAvailable:     !hasResultWithMessage(results, "sops binary not found"),
        AgeAvailable:      !hasResultWithMessage(results, "Age key file not found"),
        SopsYamlAvailable: !hasResultWithMessage(results, ".sops.yaml not found"),
    }

    // Step 5: Create and run TUI
    model := app.NewAppModel(env)
    p := tea.NewProgram(model)
    if _, err := p.Run(); err != nil {
        fmt.Fprintf(os.Stderr, "Error running sops-tui: %v\n", err)
        os.Exit(1)
    }
}

func hasResultWithMessage(results []validator.ValidationResult, substr string) bool {
    for _, r := range results {
        if strings.Contains(r.Message, substr) {
            return true
        }
    }
    return false
}
```
