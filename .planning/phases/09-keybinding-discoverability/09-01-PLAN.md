```go
func (m DetailModel) Hints() []keys.MenuHint {
    hints := keys.HintsFromBindings(m.keys.ShortHelp())
    for _, hidden := range m.keys.HiddenFromMenu() {
        for i := range hints {
            if hints[i].Mnemonic == hidden.Help().Key {
                hints[i].Visible = false
                break
            }
        }
    }
    return hints
}
